! 1. Enable subnet overlap (CRITICAL)
config system settings
    set allow-subnet-overlap enable
end

! 2. IPsec Phase 1
config vpn ipsec phase1-interface
    edit "to-CSR1000v"
        set interface "port1"
        set ike-version 2
        set peertype any
        set net-device disable
        set proposal des-sha256
        set remote-gw 192.168.20.1
        set psksecret VPNKey123
    next
end

! 3. IPsec Phase 2 (transport mode, protocol 47)
config vpn ipsec phase2-interface
    edit "to-CSR1000v"
        set phase1name "to-CSR1000v"
        set proposal des-sha256
        set dhgrp 14
        set auto-negotiate enable
        set protocol 47
        set encapsulation transport-mode
        set src-subnet 192.168.20.2 255.255.255.255
        set dst-subnet 192.168.20.1 255.255.255.255
    next
end

! 4. IPsec tunnel interface (NO IP)
config system interface
    edit "to-CSR1000v"
        set vdom "root"
        set allowaccess ping
        set type tunnel
        set interface "port1"
    next
end

! 5. GRE tunnel
config system gre-tunnel
    edit "gre-to-CSR"
        set interface "to-CSR1000v"
        set local-gw 192.168.20.2
        set remote-gw 192.168.20.1
    next
end

! 6. GRE interface with tunnel IP
config system interface
    edit "gre-to-CSR"
        set vdom "root"
        set ip 10.0.0.2 255.255.255.255
        set allowaccess ping
        set type tunnel
        set src-check disable
        set remote-ip 10.0.0.1 255.255.255.255
        set interface "to-CSR1000v"
    next
end

! 7. Static routes (CRITICAL)
config router static
    edit 1
        set dst 10.0.0.0 255.255.255.252
        set device "gre-to-CSR"
    next
    edit 3
        set dst 192.168.20.1 255.255.255.255
        set device "to-CSR1000v"
    next
end

! 8. Firewall policies
config firewall policy
    edit 1
        set name "GRE-TO-PHYSICAL"
        set srcintf "gre-to-CSR"
        set dstintf "port1"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
    next
    edit 3
        set name "Enable IPsec"
        set srcintf "to-CSR1000v"
        set dstintf "to-CSR1000v"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
    next
    edit 4
        set name "IPSEC-TO-GRE"
        set srcintf "to-CSR1000v"
        set dstintf "gre-to-CSR"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
    next
    edit 6
        set name "GRE-TO-GRE"
        set srcintf "gre-to-CSR"
        set dstintf "gre-to-CSR"
        set action accept
        set srcaddr "all"
        set dstaddr "all"
        set schedule "always"
        set service "ALL"
    next
end

config router ospf
    set router-id 2.2.2.2

    ! Define Area 0
    config area
        edit 0.0.0.0
        next
    end

    ! Advertise networks
    config network
        edit 1
            set prefix 10.0.0.0 255.255.255.252
        next
        edit 2
            set prefix 192.168.20.0 255.255.255.0
        next
    end

    ! Configure OSPF on the GRE tunnel
    config ospf-interface
        edit "gre-ospf"
            set interface "gre-to-CSR"
            set network-type point-to-point
        next
    end
end
